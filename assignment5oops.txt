ques 1: #include <iostream>
using namespace std;

class Base {
public:
    void show() {
        cout << "This is Base class" << endl;
    }
};

class Derived : public Base {
public:
    void display() {
        cout << "This is Derived class" << endl;
    }
};

int main() {
    Base b;
    Derived d;

    b.show();
    d.show();     
    d.display();

    return 0;
}

Hybrid Inheritance
#include <iostream>
using namespace std;

class A {
public:
    void showA() {
        cout << "Class A" << endl;
    }
};

class B : public A {};
class C : public A {};

class D : public B, public C {
};

int main() {
    D obj;
    obj.B::showA(); // avoid ambiguity
    obj.C::showA();
}

QUES 4 : #include <iostream>
using namespace std;

class Book {
protected:
    string title;
    string author;
    float price;

public:
    Book(string t, string a, float p) {
        title = t;
        author = a;
        price = p;
    }

    void showBook() {
        cout << "Title: " << title << endl;
        cout << "Author: " << author << endl;
        cout << "Price: " << price << endl;
    }
};

class Textbook : public Book {
    string subject;

public:
    Textbook(string t, string a, float p, string s)
        : Book(t, a, p) {
        subject = s;
    }

    void showTextbook() {
        showBook();
        cout << "Subject: " << subject << endl;
    }
};

int main() {
    Textbook tb("C++ Programming", "Bjarne Stroustrup", 599.99, "Computer Science");
    tb.showTextbook();

    return 0;
}

QUES 5 :#include <iostream>
using namespace std;

class Vehicle {
protected:
    string make;
    string model;
    int year;
public:
    Vehicle(string mk, string mdl, int yr) {
        make = mk;
        model = mdl;
        year = yr;
    }
    void displayVehicle() {
        cout << make << endl;
        cout << model << endl;
        cout << year << endl;
    }
};

class Truck : public Vehicle {
protected:
    int load_capacity;
public:
    Truck(string mk, string mdl, int yr, int load)
        : Vehicle(mk, mdl, yr) {
        load_capacity = load;
    }
    void displayTruck() {
        displayVehicle();
        cout << load_capacity << endl;
    }
};

class RefrigeratedTruck : public Truck {
    int temperature_control;
public:
    RefrigeratedTruck(string mk, string mdl, int yr, int load, int temp)
        : Truck(mk, mdl, yr, load) {
        temperature_control = temp;
    }
    void display() {
        displayTruck();
        cout << temperature_control << endl;
    }
};

int main() {
    RefrigeratedTruck r("Volvo", "FH16", 2022, 20, -5);
    r.display();
    return 0;
}

QUES  :#include <iostream>
using namespace std;

class Person {
protected:
    string name;
    string address;
public:
    Person(string n, string a) {
        name = n;
        address = a;
    }
    void showPerson() {
        cout << name << endl;
        cout << address << endl;
    }
};

class Staff : public Person {
protected:
    int employee_id;
    string department;
public:
    Staff(string n, string a, int id, string dept)
        : Person(n, a) {
        employee_id = id;
        department = dept;
    }
    void showStaff() {
        showPerson();
        cout << employee_id << endl;
        cout << department << endl;
    }
};

class Student : public Person {
protected:
    int student_id;
    string grade;
public:
    Student(string n, string a, int sid, string g)
        : Person(n, a) {
        student_id = sid;
        grade = g;
    }
    void showStudent() {
        showPerson();
        cout << student_id << endl;
        cout << grade << endl;
    }
};

class TeachingAssistant : public Staff, public Student {
public:
    TeachingAssistant(string n, string a, int id, string dept, int sid, string g)
        : Person(n, a),
          Staff(n, a, id, dept),
          Student(n, a, sid, g) {
    }
    void display() {
        Staff::showStaff();
        Student::showStudent();
    }
};

int main() {
    TeachingAssistant t("Ali", "Lahore", 101, "CS", 2023, "A");
    t.display();
    return 0;
}
